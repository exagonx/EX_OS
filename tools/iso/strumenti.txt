# Lo legge `toolinst` (cioe' anche `install -tools`) per sapere cosa c'e'
# sul CD, cosa si puo' scegliere, cosa dipende da cosa e quanto pesa.
#
# !! IL CATALOGO STA SUL CD, NON DENTRO IL PROGRAMMA. Fino ad agosto 2026 i
# gruppi erano tre liste di percorsi dentro bin/toolinst/toolinst.c: mettere
# uno strumento nuovo sul CD voleva dire ricompilare l'installatore, e finche'
# non lo si faceva quello strumento non era selezionabile - c'era e non si
# vedeva. Cosi' invece il CD si descrive da se'.
#
# !! SI DICE COSA E' SOLO DI UN GRUPPO, NON COSA VA COPIATO. Tutto quello che
# nessun blocco rivendica appartiene alla base e si copia sempre. E' la
# differenza che conta il giorno che sul CD compare un file nuovo: con un
# elenco di cose DA COPIARE resterebbe indietro in silenzio, e nessuno
# collegherebbe il link fallito di sei mesi dopo a questo file. Cosi' finisce
# sul disco insieme al resto, e l'unico modo di perderlo e' averlo scritto
# qui dentro.
#
# -----------------------------------------------------------------------------
# LE CHIAVI
#
#   nome    come si chiama, per chi installa
#   dice    cosa contiene, in una riga
#   prova   un percorso che, se c'e' sul CD, dice che il gruppo c'e' davvero.
#           Un gruppo elencato qui ma assente dal CD non viene proposto:
#           chiederlo darebbe una domanda la cui risposta non cambia niente
#   vuole   un gruppo da cui questo dipende. Sceglierlo tira dentro l'altro
#   mbyte   quanto pesa, per dire lo spazio PRIMA di cominciare a copiare
#   sempre  si -> non si puo' togliere
#   solo    un percorso, relativo alla radice dell'albero, che appartiene
#           SOLO a questo gruppo. Piu' righe = piu' percorsi
#   file    un NOME di file che appartiene solo a questo gruppo, ovunque si
#           trovi. Serve dove il percorso contiene la versione di GCC e
#           quindi non si puo' scrivere per intero
#   nota    un avvertimento da stampare a chi sceglie questo gruppo. Piu'
#           righe = piu' righe di avvertimento
#
# Le righe che non cominciano con una chiave nota vengono ignorate: un
# catalogo scritto per un toolinst piu' nuovo non deve rompere quello vecchio.
#
# -----------------------------------------------------------------------------
# RIFARE IL CD E AGGIORNARE UN DISCO
#
#   toolinst -a /disco       copia solo cio' che qui dentro e' cambiato
#
# Confronta il CONTENUTO, non le date: mkiso.py scrive in ogni file la data
# di costruzione dell'ISO, quindi dopo un `make iso` le date direbbero che
# e' cambiato tutto.
# =============================================================================

[base]
nome   = C
dice   = gcc, cpp, as, ld, la libc e gli header di sistema
prova  = bin/gcc
sempre = si
mbyte  = 50

# !! IL C NON SI SCEGLIE, e non e' una comodita': cc1plus, fbc e libcrypto
# arrivano tutti a un eseguibile passando da `as` e da `ld`. Un C++ senza
# assemblatore e' un compilatore che compila e non produce niente.


[build]
nome  = Costruzione
dice  = make, ar, ranlib, nm, strip, objcopy, objdump
prova = bin/make
vuole = base
mbyte = 22
solo  = bin/make
solo  = bin/ar
solo  = bin/ranlib
solo  = bin/nm
solo  = bin/strip
solo  = bin/objcopy
solo  = bin/objdump
solo  = prova-make

nota  = Con questo si COSTRUISCE, non si compila soltanto: make decide
nota  = cosa rifare e in che ordine, `ar` mette gli oggetti in una
nota  = libreria. E' cio' che serve a compilare un programma vero — il
nota  = primo e' FreeBASIC — invece che un file per volta.
nota  = !! La prova sta in /prova-make: si copia in una directory
nota  = scrivibile (il CD non lo e'), si lancia `make`, e `./prova` deve
nota  = dire 385.
nota  = !! `make --output-sync` NON c'e' e lo dice: vuole i lock di record
nota  = di fcntl, che EX-OS non ha. Tutto il resto di GNU make 4.2 c'e',
nota  = -j compreso.

# !! ar E RANLIB SONO DUE FILE, non uno con due nomi. Sui sistemi veri sono
# lo stesso binario chiamato in due modi; EX-OS non ha i collegamenti, e
# quindi qui sono due copie da 1,3 MB l'una. E' il grosso dei 22 MB.


[cpp]
nome  = C++
dice  = cc1plus, libstdc++ e i suoi header
prova = bin/g++
vuole = base
mbyte = 77
solo  = bin/g++
solo  = include/c++
solo  = lib/libstdc++.a

# cc1plus sta accanto a cc1 dentro libexec/gcc/<bersaglio>/<versione>, cioe'
# sotto un percorso che contiene la versione di GCC: quella cambia, il nome
# del file no.
file  = cc1plus

# !! g++ NON E' UN'ETICHETTA. Lo stesso binario decide da COME E' STATO
# CHIAMATO se compilare in C o in C++, quale cc1 lanciare e se collegare
# libstdc++. Senza g++, cc1plus c'e' e non esiste un modo di arrivarci.


[fb]
nome  = FreeBASIC
dice  = fbc, libfb e i .bi
prova = bin/fbc
vuole = base
mbyte = 2
solo  = bin/fbc
solo  = include/freebasic
solo  = lib/freebasic

nota  = `fbc prog.bas -x prog` compila, assembla e collega in un comando
nota  = solo (dal 10 agosto 2026: bersaglio `exos` nel compilatore).
nota  = !! -o e' il nome dell'OGGETTO, -x quello dell'ESEGUIBILE.
nota  = `#include "crt.bi"` funziona: da li' si chiama la libc di EX-OS.
nota  = !! I tipi sono i nostri, non quelli di Linux: time_t e' 8 byte,
nota  = CLOCKS_PER_SEC vale 100, BUFSIZ 4096.


[fbsrc]
nome  = Sorgenti FreeBASIC
dice  = i .bas del compilatore e i .c della runtime, da ricostruire qui
prova = freebasic/makefile
vuole = build
mbyte = 8
solo  = freebasic

nota  = Serve a ricostruire FreeBASIC DENTRO EX-OS: e' il programma vero
nota  = per cui esistono make, gcc, as, ld, ar e ranlib.
nota  = !! I bersagli sono `rtlib` e `compiler`, non quello predefinito:
nota  = quello include gfxlib2, che vuole X11 o il framebuffer di Linux.
nota  = !! Vanno dette le variabili del bersaglio, o il makefile chiama
nota  = `uname` e non riconosce la risposta:
nota  =     make TARGET_OS=exos TARGET_ARCH=x86 \
nota  =          CFLAGS="-O2 -DDISABLE_FFI" rtlib
nota  = Le istruzioni per esteso stanno in /freebasic/leggimi.txt.

# !! DIPENDE DA [build] E NON DA [fb]. Per RICOSTRUIRE FreeBASIC serve make;
# per USARE fbc no. Chi installa i sorgenti senza make si ritroverebbe otto
# megabyte di .bas che non puo' compilare.


[ssl]
nome  = OpenSSL
dice  = libcrypto e i suoi header
prova = lib/libcrypto.a
vuole = base
mbyte = 8
solo  = lib/libcrypto.a
solo  = include/openssl

nota  = c'e' libcrypto, NON libssl: ssl/rio/ vuole fd_set e il polling
nota  = sui socket, che qui non ci sono ancora.
nota  = RAND_bytes rifiuta su una macchina appena accesa: il serbatoio di
nota  = entropia e' 32 byte e una sola chiamata lo svuota.
