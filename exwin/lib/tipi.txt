# /exwin/lib/tipi.txt — con che cosa si apre un file (@IMMAGINI, 28 settembre 2026)
#
# Una riga per programma: le estensioni (minuscole, separate da spazi), una
# barra verticale, il programma. Il file manager lo legge al doppio clic su un
# file; cio' che non e' elencato si apre con l'editor, com'e' sempre stato.
#
# ! E' UN FILE DEL SISTEMA, NON DI UN PROGRAMMA: la regola «questa estensione
# si apre con quel programma» serve al file manager oggi, e domani alla
# scrivania e al navigatore. Scritta dentro uno di loro, gli altri la
# rifarebbero a modo loro.
#
# ! IL PROGRAMMA SI SCRIVE COME STA SU UN SISTEMA INSTALLATO (/exwin/bin/...):
# avviando dal CD chi legge questo file lo cerca anche sotto /cdrom.

# ! I GRUPPI E GLI INDIRIZZI (voce 109, 21 novembre 2026): una riga "[Nome]"
# apre un gruppo, che Impostazioni > Applicazioni mostra come scelta; un tipo
# che finisce coi due punti e' un indirizzo (http: vuol dire http://...); un
# programma vuoto vuol dire nessuno, per ora. Chi legge con exwin prima della
# 0.023 salta le righe senza barra e non chiede mai "http:": non si rompe.
#
# ! LE SCELTE DELL'UTENTE NON STANNO QUI: questo file lo riscrive netupdate.
# Impostazioni le scrive in /boot/tipi.cfg, che si legge prima di questo.

[Immagini]
bmp png jpg jpeg gif ico webp | /exwin/bin/immagini

[Testi e documenti]
txt rtf doc odt           | /exwin/bin/exeditor

[Pagine web]
html htm                  | /exwin/bin/exbrowser

[Audio]
wav mp3                   | /exwin/bin/explayer
flac ogg opus             | /exos/bin/vlcapri

[Video]
mp4 mkv webm avi mov      | /exos/bin/vlcapri

[Archivi]
zip tar gz tgz            | /exwin/bin/archivi

# ! A FOLDER IS THE TYPE "/" (voce 118, 24 November 2026): what opens a
# directory - from Firefox's "show in folder", from ex_open_file on a path
# that is a directory. A reader older than exwin 0.024 never asks for "/".
[Cartelle]
/                         | /exwin/bin/filemgr

[Indirizzi]
http: https:              | /exwin/bin/exbrowser
ftp: ftps:                |
smb:                      |
